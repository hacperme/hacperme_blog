```c
#include <stdio.h>

int test = 10;

void func(int n)
{
	int a = test;
	
	switch (n)
	{
		
		case 0: 
			int a= test;
			printf("0 a:%d\n", a);
			break;
			
		case 1:
			printf("1 a:%d\n", a);
			break;
	
	}

}

int main()
{
	func(1);
	func(0);
   
    return 0;
}
```

反编译还原大致如下

```c
int test = 10;

void func(int b)
{
    int a = test;

    if (b == 0) {
        int a = test;        // 局部遮蔽
        printf("0 a:%d\n", a);
    } else if (b == 1) {
        printf("1 a:%d\n", a); // ⚠️ 汇编中错误访问了内层 a 的偏移
    }
}

```
